cmake_minimum_required(VERSION 3.21)
if(WIN32 AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    if(DEFINED ENV{VCPKG_ROOT})
        set(_vcpkg_root "$ENV{VCPKG_ROOT}")
    else()
        set(_vswhere "$ENV{ProgramFiles\(x86\)}/Microsoft Visual Studio/Installer/vswhere.exe")
        if(EXISTS "${_vswhere}")
            execute_process(COMMAND "${_vswhere}" -latest -products * -requires Microsoft.VisualStudio.Component.VC.Tools.x86.x64 -property installationPath OUTPUT_VARIABLE _vsroot OUTPUT_STRIP_TRAILING_WHITESPACE)
            set(_vcpkg_root "${_vsroot}/VC/vcpkg")
        endif()
    endif()
    if(EXISTS "${_vcpkg_root}/scripts/buildsystems/vcpkg.cmake")
        set(CMAKE_TOOLCHAIN_FILE "${_vcpkg_root}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "vcpkg")
    else()
        message(FATAL_ERROR "vcpkg не знайдено. Установіть компонент vcpkg у Visual Studio або задайте VCPKG_ROOT.")
    endif()
endif()
project(TPP_Lab7 LANGUAGES CXX)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LAB7_INJECT_BUG "Навмисна помилка для негативного самоконтролю" OFF)
add_library(MatrixLib STATIC src/MatrixAnalysis.cpp src/Matrix.cpp)
target_include_directories(MatrixLib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(LAB7_INJECT_BUG)
    target_compile_definitions(MatrixLib PRIVATE LAB7_INJECT_BUG)
endif()
if(MSVC)
    target_compile_options(MatrixLib PRIVATE /W4 /utf-8)
else()
    target_compile_options(MatrixLib PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(MatrixDemo demo/main.cpp)
target_link_libraries(MatrixDemo PRIVATE MatrixLib)
add_executable(AntipatternDemo antipattern/main.cpp)
find_package(GTest CONFIG REQUIRED)
enable_testing()
add_executable(MatrixTests tests/MatrixTests.cpp)
target_link_libraries(MatrixTests PRIVATE MatrixLib GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(MatrixTests)
